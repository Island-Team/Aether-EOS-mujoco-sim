cmake_minimum_required(VERSION 3.22)
project(eos_mujoco_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf REQUIRED)
find_package(gRPC REQUIRED)
find_package(Threads REQUIRED)
find_package(JPEG REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
find_library(EGL_LIBRARY EGL REQUIRED)
set(MUJOCO_ROOT "$ENV{MUJOCO_ROOT}" CACHE PATH "MuJoCo installation root")
if(NOT MUJOCO_ROOT)
  message(FATAL_ERROR "Set MUJOCO_ROOT to the directory containing include/mujoco and lib/libmujoco.so")
endif()
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h PATHS "${MUJOCO_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(MUJOCO_LIBRARY mujoco PATHS "${MUJOCO_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(mujoco SHARED IMPORTED)
set_target_properties(mujoco PROPERTIES IMPORTED_LOCATION "${MUJOCO_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MUJOCO_INCLUDE_DIR}")
set(PROTO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/proto")
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")
foreach(NAME simulator benchmark)
  add_custom_command(OUTPUT "${GENERATED_DIR}/${NAME}.pb.cc" "${GENERATED_DIR}/${NAME}.pb.h" "${GENERATED_DIR}/${NAME}.grpc.pb.cc" "${GENERATED_DIR}/${NAME}.grpc.pb.h"
    COMMAND protobuf::protoc --proto_path=${PROTO_DIR} --cpp_out=${GENERATED_DIR} --grpc_out=${GENERATED_DIR} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> "${PROTO_DIR}/${NAME}.proto"
    DEPENDS "${PROTO_DIR}/${NAME}.proto" VERBATIM)
  list(APPEND GENERATED_SOURCES "${GENERATED_DIR}/${NAME}.pb.cc" "${GENERATED_DIR}/${NAME}.grpc.pb.cc")
endforeach()
add_custom_target(proto_generated DEPENDS ${GENERATED_SOURCES})
add_executable(eos-mujoco-sim src/main.cpp src/service.cpp src/engine.cpp ${GENERATED_SOURCES})
add_dependencies(eos-mujoco-sim proto_generated)
target_include_directories(eos-mujoco-sim PRIVATE "${GENERATED_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(eos-mujoco-sim PRIVATE gRPC::grpc++ protobuf::libprotobuf Threads::Threads JPEG::JPEG ${EGL_LIBRARY} mujoco)
set_target_properties(eos-mujoco-sim PROPERTIES BUILD_RPATH "${MUJOCO_ROOT}/lib")
enable_testing()
add_executable(smoke-rpc tests/smoke.cpp ${GENERATED_SOURCES})
add_dependencies(smoke-rpc proto_generated)
target_include_directories(smoke-rpc PRIVATE "${GENERATED_DIR}")
target_link_libraries(smoke-rpc PRIVATE gRPC::grpc++ protobuf::libprotobuf)
add_executable(benchmark-eval tools/benchmark-eval.cpp ${GENERATED_SOURCES})
add_dependencies(benchmark-eval proto_generated)
target_include_directories(benchmark-eval PRIVATE "${GENERATED_DIR}")
target_link_libraries(benchmark-eval PRIVATE gRPC::grpc++ protobuf::libprotobuf)
add_test(NAME smoke-rpc COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.sh" $<TARGET_FILE:eos-mujoco-sim> "${CMAKE_CURRENT_SOURCE_DIR}/assets/scene.xml" $<TARGET_FILE:smoke-rpc>)
add_test(NAME smoke-rpc-navigation COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.sh" $<TARGET_FILE:eos-mujoco-sim> "${CMAKE_CURRENT_SOURCE_DIR}/assets/scenes/navigation/scene.xml" $<TARGET_FILE:smoke-rpc>)
add_test(NAME benchmark-eval-pick-place COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/eval.py" $<TARGET_FILE:eos-mujoco-sim> $<TARGET_FILE:benchmark-eval> "${CMAKE_CURRENT_SOURCE_DIR}/assets/scene.xml" pick-place)
add_test(NAME benchmark-eval-navigation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/eval.py" $<TARGET_FILE:eos-mujoco-sim> $<TARGET_FILE:benchmark-eval> "${CMAKE_CURRENT_SOURCE_DIR}/assets/scenes/navigation/scene.xml" navigation)
add_executable(engine-test tests/engine.cpp src/engine.cpp)
target_include_directories(engine-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(engine-test PRIVATE Threads::Threads JPEG::JPEG ${EGL_LIBRARY} mujoco)
set_target_properties(engine-test PROPERTIES BUILD_RPATH "${MUJOCO_ROOT}/lib")
add_test(NAME engine-test COMMAND $<TARGET_FILE:engine-test> "${CMAKE_CURRENT_SOURCE_DIR}/assets/scene.xml" "${CMAKE_CURRENT_SOURCE_DIR}/assets/scenes/navigation/scene.xml")
add_test(NAME scene-view-check COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tools/scene-view.sh" --check)
set_tests_properties(scene-view-check PROPERTIES ENVIRONMENT "MUJOCO_ROOT=${MUJOCO_ROOT}")
add_test(NAME scene-view-check-navigation COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tools/scene-view.sh" --check --scene "${CMAKE_CURRENT_SOURCE_DIR}/assets/scenes/navigation/scene.xml")
set_tests_properties(scene-view-check-navigation PROPERTIES ENVIRONMENT "MUJOCO_ROOT=${MUJOCO_ROOT}")
